cmake_minimum_required(VERSION 3.15)
project(gam8_libretro LANGUAGES C)
option(GAM8_STATIC "Build an archive for statically linked frontends" OFF)
if(GAM8_STATIC)
  set(core_type STATIC)
else()
  set(core_type SHARED)
endif()
add_library(gam8_libretro ${core_type} src/core.c src/machine.c src/psg.c)
set_target_properties(gam8_libretro PROPERTIES PREFIX "" C_STANDARD 11 C_STANDARD_REQUIRED YES C_VISIBILITY_PRESET hidden)
install(TARGETS gam8_libretro RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION .)
install(FILES gam8_libretro.info LICENSE NOTICE DESTINATION .)
install(FILES third_party/MAME-BSD.txt DESTINATION third_party)
option(GAM8_TESTS "Build local diagnostic executables" OFF)
if(GAM8_TESTS)
  enable_testing()
  add_executable(gam8_devices tests/devices.c src/psg.c)
  add_executable(gam8_presentation tests/presentation.c src/machine.c src/psg.c)
  add_executable(gam8_frontend tests/frontend.c tests/traced_machine.c src/core.c src/psg.c)
  set_target_properties(gam8_devices gam8_presentation gam8_frontend
                        PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
  add_test(NAME gam8_devices COMMAND gam8_devices)
  add_test(NAME gam8_presentation COMMAND gam8_presentation)
endif()
