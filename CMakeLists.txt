cmake_minimum_required(VERSION 3.15)
project(gamate_libretro LANGUAGES C)
option(GAMATE_STATIC "Build an archive for statically linked frontends" OFF)
if(GAMATE_STATIC)
  set(core_type STATIC)
else()
  set(core_type SHARED)
endif()
add_library(gamate_libretro ${core_type} src/core.c src/machine.c src/psg.c)
set_target_properties(gamate_libretro PROPERTIES PREFIX "" C_STANDARD 11 C_STANDARD_REQUIRED YES C_VISIBILITY_PRESET hidden)
option(GAMATE_TESTS "Build local diagnostic executables" OFF)
if(GAMATE_TESTS)
  enable_testing()
  add_executable(gamate_devices tests/devices.c src/psg.c)
  add_executable(gamate_presentation tests/presentation.c src/machine.c src/psg.c)
  add_executable(gamate_frontend tests/frontend.c tests/traced_machine.c src/core.c src/psg.c)
  set_target_properties(gamate_devices gamate_presentation gamate_frontend
                        PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
  add_test(NAME gamate_devices COMMAND gamate_devices)
  add_test(NAME gamate_presentation COMMAND gamate_presentation)
endif()
